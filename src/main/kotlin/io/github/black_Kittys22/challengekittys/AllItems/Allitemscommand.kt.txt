package io.github.black_Kittys22.challengekittys.AllItems

import io.github.black_Kittys22.challengekittys.Main
import net.kyori.adventure.text.Component
import net.kyori.adventure.text.format.NamedTextColor
import net.kyori.adventure.text.format.TextDecoration
import org.bukkit.Bukkit
import org.bukkit.Material
import org.bukkit.Sound
import org.bukkit.command.Command
import org.bukkit.command.CommandExecutor
import org.bukkit.command.CommandSender
import org.bukkit.entity.Player
import org.bukkit.event.EventHandler
import org.bukkit.event.Listener
import org.bukkit.event.inventory.InventoryClickEvent
import org.bukkit.event.inventory.InventoryDragEvent
import org.bukkit.event.inventory.InventoryCloseEvent
import org.bukkit.inventory.ItemStack
import java.util.UUID

class AllItemsCommand(private val plugin: Main) : CommandExecutor, Listener {

    private val playerPages = mutableMapOf<UUID, Int>()

    override fun onCommand(sender: CommandSender, command: Command, label: String, args: Array<out String>): Boolean {
        if (sender !is Player) {
            sender.sendMessage(Component.text("Dieser Befehl ist nur für Spieler!", NamedTextColor.RED))
            return true
        }

        if (!plugin.isAllItemsChallengeActive) {
            sender.sendMessage(Component.text("Die All Items Challenge ist nicht aktiv!", NamedTextColor.RED))
            return true
        }

        if (args.isEmpty()) {
            showProgress(sender)
        } else {
            when (args[0].lowercase()) {
                "progress", "fortschritt" -> showProgress(sender)
                "gui", "list", "liste" -> {
                    playerPages[sender.uniqueId] = 0
                    openProgressGUI(sender, 0)
                }
                "missing", "fehlend" -> showMissing(sender)
                else -> {
                    sender.sendMessage(Component.text("Verwendung:", NamedTextColor.YELLOW))
                    sender.sendMessage(Component.text("  /allitems - Zeigt deinen Fortschritt", NamedTextColor.GRAY))
                    sender.sendMessage(Component.text("  /allitems gui - Öffnet die Item-Liste", NamedTextColor.GRAY))
                    sender.sendMessage(Component.text("  /allitems missing - Zeigt fehlende Items", NamedTextColor.GRAY))
                }
            }
        }
        return true
    }

    private fun showProgress(player: Player) {
        val listener = plugin.allItemsListener
        val (collected, total) = listener.getProgress(player.name)
        val percentage = if (total > 0) (collected * 100.0 / total).toInt() else 0

        player.sendMessage(Component.text("━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━", NamedTextColor.GOLD, TextDecoration.BOLD))
        player.sendMessage(Component.text("  📦 All Items Challenge (TEAM)", NamedTextColor.YELLOW, TextDecoration.BOLD))
        player.sendMessage(Component.text("━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━", NamedTextColor.GOLD, TextDecoration.BOLD))

        val progressBar = createProgressBar(percentage)
        player.sendMessage(
            Component.text("  Team-Fortschritt: ", NamedTextColor.GRAY)
                .append(Component.text("$collected", NamedTextColor.GREEN, TextDecoration.BOLD))
                .append(Component.text(" / ", NamedTextColor.GRAY))
                .append(Component.text("$total", NamedTextColor.GOLD))
        )
        player.sendMessage(Component.text("  ").append(progressBar))
        player.sendMessage(
            Component.text("  ", NamedTextColor.GRAY)
                .append(Component.text("$percentage%", getColorForPercentage(percentage), TextDecoration.BOLD))
                .append(Component.text(" abgeschlossen", NamedTextColor.GRAY))
        )
        player.sendMessage(Component.text("━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━", NamedTextColor.GOLD, TextDecoration.BOLD))
    }

    private fun createProgressBar(percentage: Int): Component {
        val barLength = 20
        val filled = (percentage / 5).coerceIn(0, barLength)
        val empty = barLength - filled
        val bar = Component.text().append(Component.text("[", NamedTextColor.DARK_GRAY))
        for (i in 0 until filled) bar.append(Component.text("█", getColorForPercentage(percentage)))
        for (i in 0 until empty) bar.append(Component.text("█", NamedTextColor.DARK_GRAY))
        bar.append(Component.text("]", NamedTextColor.DARK_GRAY))
        return bar.build()
    }

    private fun getColorForPercentage(percentage: Int): NamedTextColor {
        return when {
            percentage >= 90 -> NamedTextColor.DARK_GREEN
            percentage >= 75 -> NamedTextColor.GREEN
            percentage >= 50 -> NamedTextColor.YELLOW
            percentage >= 25 -> NamedTextColor.GOLD
            else -> NamedTextColor.RED
        }
    }

    private fun showMissing(player: Player) {
        val listener = plugin.allItemsListener
        val missing = listener.getMissingItems(player.name).sortedBy { it.name }
        if (missing.isEmpty()) {
            player.sendMessage(Component.text("✓ Das Team hat alle Items gesammelt!", NamedTextColor.GREEN, TextDecoration.BOLD))
            return
        }
        player.sendMessage(Component.text("━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━", NamedTextColor.RED, TextDecoration.BOLD))
        player.sendMessage(Component.text("  Fehlende Items (TEAM): ${missing.size}", NamedTextColor.YELLOW, TextDecoration.BOLD))
        player.sendMessage(Component.text("━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━", NamedTextColor.RED, TextDecoration.BOLD))
        missing.take(10).forEach { material ->
            player.sendMessage(Component.text("  ✗ ", NamedTextColor.RED).append(Component.text(material.name.replace("_", " "), NamedTextColor.GRAY)))
        }
        if (missing.size > 10) player.sendMessage(Component.text("  ... und ${missing.size - 10} weitere", NamedTextColor.DARK_GRAY, TextDecoration.ITALIC))
        player.sendMessage(Component.text("━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━", NamedTextColor.RED, TextDecoration.BOLD))
    }

    private fun openProgressGUI(player: Player, page: Int) {
        val listener = plugin.allItemsListener
        val allItems = listener.getAllRequiredItems().sortedBy { it.name }
        val itemsPerPage = 36
        val totalPages = Math.max(1, (allItems.size + itemsPerPage - 1) / itemsPerPage)
        val currentPage = page.coerceIn(0, totalPages - 1)

        playerPages[player.uniqueId] = currentPage

        val title = Component.text("All Items - Seite ${currentPage + 1}/$totalPages", NamedTextColor.GOLD, TextDecoration.BOLD)
        val inv = Bukkit.createInventory(null, 54, title)

        // Info Item Slot 4
        val (collectedCount, total) = listener.getProgress(player.name)
        val infoItem = ItemStack(Material.BOOK)
        val infoMeta = infoItem.itemMeta
        infoMeta.displayName(Component.text("Team-Fortschritt", NamedTextColor.GOLD, TextDecoration.BOLD).decoration(TextDecoration.ITALIC, false))
        infoMeta.lore(listOf(
            Component.text("Gesammelt: $collectedCount / $total", NamedTextColor.GRAY).decoration(TextDecoration.ITALIC, false),
            Component.text("Seite: ${currentPage + 1} von $totalPages", NamedTextColor.YELLOW).decoration(TextDecoration.ITALIC, false)
        ))
        infoItem.itemMeta = infoMeta
        inv.setItem(4, infoItem)

        // Glass Füllung oben
        val glassPane = ItemStack(Material.GRAY_STAINED_GLASS_PANE)
        val glassMeta = glassPane.itemMeta
        glassMeta.displayName(Component.text(" "))
        glassPane.itemMeta = glassMeta
        for (i in 0..8) if (i != 4) inv.setItem(i, glassPane)

        // Items laden
        val startIndex = currentPage * itemsPerPage
        val endIndex = minOf(startIndex + itemsPerPage, allItems.size)
        val pageItems = allItems.subList(startIndex, endIndex)
        val collected = listener.getCollectedItems(player.name)

        var slot = 9
        pageItems.forEach { material ->
            val isCollected = collected.contains(material)
            val item = ItemStack(material)
            val meta = item.itemMeta
            val color = if (isCollected) NamedTextColor.GREEN else NamedTextColor.RED
            meta.displayName(Component.text(material.name.replace("_", " "), color).decoration(TextDecoration.ITALIC, false))
            meta.lore(listOf(Component.text(if (isCollected) "✓ Gesammelt" else "❌ Noch nicht gesammelt", color).decoration(TextDecoration.ITALIC, false)))
            item.itemMeta = meta
            inv.setItem(slot++, item)
        }

        // Navigation
        if (currentPage > 0) {
            val prev = ItemStack(Material.ARROW)
            val pMeta = prev.itemMeta
            pMeta.displayName(Component.text("← Vorherige Seite", NamedTextColor.YELLOW))
            prev.itemMeta = pMeta
            inv.setItem(45, prev)
        }

        val close = ItemStack(Material.BARRIER)
        val cMeta = close.itemMeta
        cMeta.displayName(Component.text("Schließen", NamedTextColor.RED))
        close.itemMeta = cMeta
        inv.setItem(49, close)

        if (currentPage < totalPages - 1) {
            val next = ItemStack(Material.ARROW)
            val nMeta = next.itemMeta
            nMeta.displayName(Component.text("Nächste Seite →", NamedTextColor.YELLOW))
            next.itemMeta = nMeta
            inv.setItem(53, next)
        }

        // Rest mit Glas füllen (WICHTIG: NACH den Pfeilen)
        for (i in 45..53) if (inv.getItem(i) == null) inv.setItem(i, glassPane)

        player.openInventory(inv)
    }

    @EventHandler
    fun onInventoryClick(event: InventoryClickEvent) {
        val player = event.whoClicked as? Player ?: return
        if (!playerPages.containsKey(player.uniqueId)) return

        event.isCancelled = true
        val slot = event.slot
        val currentPage = playerPages[player.uniqueId] ?: 0

        // Navigation Logik
        when (slot) {
            45 -> if (currentPage > 0) {
                openProgressGUI(player, currentPage - 1)
                player.playSound(player.location, Sound.UI_BUTTON_CLICK, 0.5f, 1.0f)
            }
            49 -> {
                player.closeInventory()
                player.playSound(player.location, Sound.UI_BUTTON_CLICK, 0.5f, 0.8f)
            }
            53 -> {
                val listener = plugin.allItemsListener
                val totalItems = listener.getAllRequiredItems().size
                val totalPages = Math.max(1, (totalItems + 35) / 36)

                if (currentPage < totalPages - 1) {
                    openProgressGUI(player, currentPage + 1)
                    player.playSound(player.location, Sound.UI_BUTTON_CLICK, 0.5f, 1.0f)
                }
            }
            in 9..44 -> {
                val clickedItem = event.currentItem ?: return
                if (clickedItem.type == Material.GRAY_STAINED_GLASS_PANE || clickedItem.type == Material.AIR) return

                if (plugin.allItemsListener.markItemAsCollected(clickedItem.type, player)) {
                    player.playSound(player.location, Sound.ENTITY_EXPERIENCE_ORB_PICKUP, 0.7f, 1.2f)
                    openProgressGUI(player, currentPage)
                }
            }
        }
    }

    @EventHandler
    fun onInventoryDrag(event: InventoryDragEvent) {
        if (playerPages.containsKey(event.whoClicked.uniqueId)) event.isCancelled = true
    }

    @EventHandler
    fun onInventoryClose(event: InventoryCloseEvent) {
        playerPages.remove(event.player.uniqueId)
    }
}
